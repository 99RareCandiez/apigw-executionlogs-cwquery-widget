fields @timestamp, @message
| parse @message /^\((?<request_id>[^)]+)\)/
| filter ispresent(request_id)
| parse @message /HTTP Method: (?<http_method>\w+), Resource Path: (?<resource_path>\S+)/
| parse @message /Api Key: (?<api_key>\S*)\s*Api Stage: (?<api_stage>\S+)/
| parse @message /Status: (?<status>\d+), Time to Response Headers: (?<ttfb_ms>\d+) ms/
| parse @message /Total Payload Bytes Received: (?<payload_bytes>\d+), Integration Latency: (?<integration_latency_ms>\d+) ms/
| stats min(@timestamp) as request_start,
        max(http_method) as http_method,
        max(resource_path) as resource_path,
        max(api_key) as api_key,
        max(api_stage) as api_stage,
        max(status) as status,
        max(ttfb_ms) as time_to_response_headers_ms,
        max(payload_bytes) as payload_bytes,
        max(integration_latency_ms) as integration_latency_ms
  by request_id
| sort request_start desc
| limit 200




fields @timestamp, @message
| filter @message like /HTTP Method:|Verifying Usage Plan|Received response headers|Recieved response headers|Response stream completed/
| parse @message /^\((?<RequestId>[^)]+)\)/
| parse @message /HTTP Method: (?<HttpMethod>[A-Z]+), Resource Path: (?<ResourcePath>[^\r\n]+)/
| parse @message /Verifying Usage Plan for request: .*?\. Api Key: ?(?<ApiKey>.*?) Api Stage: (?<ApiStage>[^\s]+)/
| parse @message /(?:Received|Recieved) response headers\. Status: (?<Status>\d+), Time to Response Headers: (?<ResponseHeaderMs>\d+) ms/
| parse @message /Response stream completed\. Total Payload Bytes Received: (?<PayloadBytes>\d+), Integration Latency: (?<IntegrationLatencyMs>\d+) ms/
| filter ispresent(RequestId)
| stats
    latest(@timestamp) as LastEvent,
    latest(HttpMethod) as Method,
    latest(ResourcePath) as ResourcePath,
    latest(ApiKey) as ApiKey,
    latest(ApiStage) as ApiStage,
    latest(Status) as Status,
    max(ResponseHeaderMs) as ResponseHeaderMs,
    max(PayloadBytes) as PayloadBytes,
    max(IntegrationLatencyMs) as IntegrationLatencyMs
  by RequestId
| sort LastEvent desc
| limit 100




fields @timestamp, @message
# 1. Extract request_id from the beginning of each line
| parse @message "(*) *" as requestId, logPayload

# 2. Extract Method and Resource Path
| parse logPayload "HTTP Method: *, Resource Path: *" as httpMethod, resourcePath

# 3. Extract API Key and Stage
| parse logPayload "Verifying Usage Plan for request: *. Api Key: *. Api Stage: *" as usageReqId, apiKey, apiStage

# 4. Extract Status Code and Time to Response Headers
| parse logPayload "Received response headers. Status: *, Time to Response Headers: * ms" as statusCode, responseHeaderTimeMs

# 5. Extract Payload Size and Integration Latency
| parse logPayload "Response stream completed. Total Payload Bytes Received: *, Integration Latency: * ms" as payloadBytes, integrationLatencyMs

# 6. Aggregate fields across all log events for each request
| stats 
    latest(httpMethod) as method,
    latest(resourcePath) as path,
    latest(apiKey) as key,
    latest(apiStage) as stage,
    latest(statusCode) as status,
    latest(responseHeaderTimeMs) as timeToHeadersMs,
    latest(payloadBytes) as bytesReceived,
    latest(integrationLatencyMs) as latencyMs
  by requestId
| filter ispresent(method) or ispresent(status)
| sort latencyMs desc

#adjusting more friendly column names. also more logic to show if buffered route instead of leaving some fields blank if buffered as opposed to streaming.
fields @timestamp, @message
| parse @message /^\((?<reqid>[^)]+)\)/
| filter ispresent(reqid)
| parse @message /HTTP Method: (?<raw_method>\w+), Resource Path: (?<raw_path>\S+)/
| parse @message /(?i)Api Stage: (?<raw_stage>\S+)/
| parse @message /(?i)Api Key: (?<raw_key>\S+) Api Stage:/
| parse @message /(?i)Received response.*?Status: (?<raw_status>\d+)/
| parse @message /(?i)Time to Response Headers: (?<raw_ttfb>\d+) ms/
| parse @message /(?i)Total Payload Bytes Rec[ei]+ved: (?<raw_payload>\d+)/
| parse @message /(?i)Integration latency: (?<raw_latency>\d+) ms/
| parse @message /(?<unauth_msg>Unauthorized request)/
| parse @message /(?<authorizer_msg>Starting authorizer)/
| stats min(@timestamp) as request_time,
        earliest(raw_method) as http_method,
        earliest(raw_path) as resource_path,
        earliest(raw_key) as api_key,
        earliest(raw_stage) as api_stage,
        max(raw_status) as status_code,
        earliest(raw_ttfb) as ttfb_text,
        earliest(raw_payload) as payload_text,
        max(raw_latency) as integration_latency_ms,
        count(raw_ttfb) as stream_lines,
        count(raw_latency) as latency_lines,
        count(unauth_msg) as unauth_count,
        count(authorizer_msg) as authorizer_runs
  by reqid
| fields if(unauth_count > 0, "yes", "no") as unauthorized,
         if(authorizer_runs > 0, "yes", "no") as authorizer_invoked,
         if(stream_lines > 0, "streaming", if(latency_lines > 0, "buffered", "none")) as integration_type
| fields if(integration_type = "buffered", "buffered", payload_text) as payload_bytes,
         if(integration_type = "buffered", "buffered", ttfb_text) as time_to_headers_ms
| sort request_time desc
| display request_time, reqid, unauthorized, authorizer_invoked, integration_type,
          http_method, resource_path, api_key, api_stage,
          status_code, time_to_headers_ms, payload_bytes, integration_latency_ms
| limit 200
